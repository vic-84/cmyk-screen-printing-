# Conexión de la señal de movimiento de filas
class AspectRatioPixmapLabel(QtWidgets.QLabel):
    """
    Un QLabel que se expande para llenar el espacio y dibuja su pixmap
    manteniendo la proporción de aspecto del contenido.
    """
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._pixmap = QtGui.QPixmap()
        # Le decimos que su política es expandirse en ambas direcciones.
        self.setSizePolicy(QtWidgets.QSizePolicy.Expanding, QtWidgets.QSizePolicy.Expanding)
        self.setMinimumSize(150, 150)
        self.setAlignment(QtCore.Qt.AlignCenter)

    def setPixmap(self, pixmap):
        super().clear()
        if isinstance(pixmap, QtGui.QPixmap) and not pixmap.isNull():
            self._pixmap = pixmap
        else:
            self._pixmap = QtGui.QPixmap()
            if len(self.text()) > 0:
                 super().setText(self.text())
        self.update()

    def paintEvent(self, event):
        """
        Toma el pixmap guardado (_pixmap), lo escala al tamaño actual del widget
        y lo dibuja centrado. VERSIÓN CORREGIDA Y ROBUSTA.
        """
        # Primero, es buena práctica llamar al paintEvent del padre
        super().paintEvent(event)

        # Si no tenemos un pixmap guardado o si es nulo, no hay nada que dibujar.
        if not hasattr(self, '_pixmap') or self._pixmap.isNull():
            return

        painter = QtGui.QPainter(self)
        # Pistas para que el escalado se vea suave
        painter.setRenderHint(QtGui.QPainter.Antialiasing)
        painter.setRenderHint(QtGui.QPainter.SmoothPixmapTransform)

        # Escala el pixmap para que quepa dentro del tamaño actual del widget, manteniendo la proporción
        # self.size() devuelve el tamaño actual del widget en la ventana
        scaled_pixmap = self._pixmap.scaled(self.size(), QtCore.Qt.KeepAspectRatio, QtCore.Qt.SmoothTransformation)

        # Calcula las coordenadas x, y para que la imagen escalada quede centrada
        x = (self.width() - scaled_pixmap.width()) / 2
        y = (self.height() - scaled_pixmap.height()) / 2

        # Dibuja el pixmap en las coordenadas calculadas
        painter.drawPixmap(int(x), int(y), scaled_pixmap)

# Clase para la etiqueta de la vista previa que detecta clics
class ClickablePreviewLabel(AspectRatioPixmapLabel):
    """
    Una subclase de AspectRatioPixmapLabel que emite una señal
    cuando el usuario hace clic en ella, pasando las coordenadas del clic.
    """
    # Se define una nueva señal que enviará la posición del clic
    clicked = QtCore.pyqtSignal(QtCore.QPoint)

    def mousePressEvent(self, event):
        # Cuando se presiona el ratón, emite la señal con la posición
        self.clicked.emit(event.pos())
        super().mousePressEvent(event)

# Clase para el visor con Zoom y Pan
class ZoomableGraphicsView(QGraphicsView):
    """
    Un QGraphicsView que permite hacer zoom con la rueda del ratón y
    panear (arrastrar) con el botón del medio o con el clic izquierdo presionado.
    """
    def __init__(self, scene, parent=None):
        super().__init__(scene, parent)

        # --- ¡CORRECCIÓN DEFINITIVA! DESACTIVAMOS TODO EL SUAVIZADO ---
        # Forzamos al visor a dibujar píxeles puros y nítidos.
        # Esto eliminará los colores "enlazados" y el ruido visual.
        self.setRenderHint(QPainter.Antialiasing, False)
        self.setRenderHint(QPainter.SmoothPixmapTransform, False)
        # --- FIN DE LA CORRECCIÓN ---

        self.setDragMode(QGraphicsView.ScrollHandDrag)
        self.setTransformationAnchor(QGraphicsView.AnchorViewCenter)
        self.setResizeAnchor(QGraphicsView.AnchorViewCenter)
        
    def wheelEvent(self, event):
        # Maneja el evento de la rueda del ratón para hacer zoom
        zoom_in_factor = 1.15
        zoom_out_factor = 1 / zoom_in_factor

        if event.angleDelta().y() > 0:
            # Rueda hacia adelante (zoom in)
            self.scale(zoom_in_factor, zoom_in_factor)
        else:
            # Rueda hacia atrás (zoom out)
            self.scale(zoom_out_factor, zoom_out_factor)

        self.setDragDropMode(QtWidgets.QAbstractItemView.InternalMove)
        self.setSelectionMode(QtWidgets.QAbstractItemView.ExtendedSelection)
        # El tamaño se ajusta al contenido
        self.setSizePolicy(QtWidgets.QSizePolicy.Minimum, QtWidgets.QSizePolicy.Expanding)

class AspectRatioContainer(QtWidgets.QWidget):
    """
    Un QWidget que coopera con los layouts de Qt para mantener una
    proporción de aspecto (alto/ancho) mientras se expande.
    """
    def __init__(self, parent=None, ratio=1.0):
        super().__init__(parent)
        self._ratio = ratio
        # Política de tamaño que le dice al layout que queremos crecer
        # y que nuestra altura preferida depende del ancho.
        self.setSizePolicy(QtWidgets.QSizePolicy.Expanding, QtWidgets.QSizePolicy.Preferred)

    def set_aspect_ratio(self, ratio):
        """Establece la nueva proporción de aspecto."""
        if ratio > 0 and self._ratio != ratio:
            self._ratio = ratio
            # Informa al layout que nuestras pistas de tamaño han cambiado.
            self.updateGeometry()

    def heightForWidth(self, width):
        """Devuelve la altura ideal para un ancho dado."""
        return int(width * self._ratio)

    def hasHeightForWidth(self):
        """Informa al sistema de layout que nuestra altura depende del ancho."""
        return True

    def update_composite_preview_fast(self):
        """
        Crea la vista previa compuesta usando ÚNICAMENTE los datos de la caché de la miniatura.
        VERSIÓN FINAL, OPTIMIZADA Y CORREGIDA.
        """
        try:
            # Verificar que la caché de la miniatura exista y no esté vacía
            if not hasattr(self, 'preview_cache_thumb') or not self.preview_cache_thumb:
                return

            print("DEBUG: Iniciando actualización OPTIMIZADA de la composición...")

            # --- INICIO DE LA CORRECCIÓN ---

            # 1. Obtiene el orden de los canales de la UI
            order = self.get_channel_order_from_ui()
            channels_to_process = [ch for ch in order if ch in self.preview_cache_thumb]

            # Si no hay canales para procesar, no hacer nada
            if not channels_to_process:
                return

            #    Esto garantiza que el lienzo SIEMPRE tendrá el mismo tamaño que los datos.
            first_channel_data = self.preview_cache_thumb[channels_to_process[0]]
            h, w = first_channel_data.shape[:2]

            # 3. Crea el canvas con las dimensiones CORRECTAS y DINÁMICAS
            canvas = np.ones((h, w, 3), dtype=np.float32)
            print(f"DEBUG: Canvas creado con tamaño dinámico y correcto: {h}x{w}")

            # 4. El bucle ahora usa la lista pre-filtrada 'channels_to_process'
            for channel in channels_to_process:
                
                # Lee los datos de la CACHÉ DE MINIATURA
                source_data = self.preview_cache_thumb[channel]
                
                # El resto de tu lógica de mezclado está perfecta y no necesita cambios
                mask = (255 - source_data).astype(np.float32) / 255.0
                ink_qcolor = self.channel_colors.get(channel, QtGui.QColor(0,0,0))
                ink_rgb = np.array([ink_qcolor.red(), ink_qcolor.green(), ink_qcolor.blue()], dtype=np.float32) / 255.0

                if channel == 'W':
                    alpha = mask[..., np.newaxis] * 1.0
                    canvas = canvas * (1.0 - alpha) + ink_rgb * alpha
                else:
                    opacity = 0.90
                    if channel == 'K': opacity = 0.98
                    alpha = mask[..., np.newaxis] * opacity
                    canvas = canvas * (1 - (alpha * (1 - ink_rgb)))

            # 5. Convierte el canvas final a QPixmap y lo muestra
            final_image = np.clip(canvas * 255, 0, 255).astype(np.uint8)
            pixmap = self.convertir_cv_a_pixmap(final_image)
            
            if not pixmap.isNull():
                self.preview_label.setPixmap(pixmap)
                print("DEBUG: Vista previa RÁPIDA y OPTIMIZADA actualizada con éxito.")
            else:
                print("WARN: El pixmap generado para la vista previa rápida es nulo.")

        except Exception as e:
            print(f"❌ Error en update_composite_preview_fast: {e}")
            import traceback
            traceback.print_exc()

    def convertir_cv_a_pixmap(self, cv_img):
        """
        Convierte una imagen de formato OpenCV (array NumPy BGR) a un QPixmap de Qt.
        """
        try:
            if cv_img is None:
                return QtGui.QPixmap()

            # Convierte la imagen de BGR (formato de OpenCV) a RGB
            rgb_image = cv2.cvtColor(cv_img, cv2.COLOR_BGR2RGB)
            
            h, w, ch = rgb_image.shape
            bytes_per_line = ch * w
            
            # Crea un QImage desde los datos del array
            q_image = QtGui.QImage(rgb_image.data, w, h, bytes_per_line, QtGui.QImage.Format_RGB888)
            
            if q_image.isNull():
                print("WARN: QImage creado a partir de datos de OpenCV es nulo.")
                return QtGui.QPixmap()
                
            # Crea un QPixmap desde el QImage
            return QtGui.QPixmap.fromImage(q_image)
            
        except Exception as e:
            print(f"❌ Error en convertir_cv_a_pixmap: {e}")
            return QtGui.QPixmap() # Devuelve un pixmap vacío en caso de error

    def detect_image_resolution(self, file_path):
        """Detectar resolución y características de la imagen"""
        try:
            # Información básica con OpenCV
            h, w = self.image.shape[:2]
            file_size = os.path.getsize(file_path) / (1024 * 1024)  # MB

            # Intentar obtener DPI con PIL
            dpi_x, dpi_y = 72, 72  # Valores por defecto
            try:
                with Image.open(file_path) as pil_img:
                    if hasattr(pil_img, 'info') and 'dpi' in pil_img.info:
                        dpi_x, dpi_y = pil_img.info['dpi']
                    elif hasattr(pil_img, 'info') and 'resolution' in pil_img.info:
                        dpi_x, dpi_y = pil_img.info['resolution']
            except:
                pass

            # Calcular tamaño de impresión en cm
            if dpi_x > 0 and dpi_y > 0:
                width_cm = (w / dpi_x) * 2.54
                height_cm = (h / dpi_y) * 2.54
            else:
                # Asumir 300 DPI si no se puede detectar
                width_cm = (w / 300) * 2.54
                height_cm = (h / 300) * 2.54
                dpi_x = dpi_y = 300

            # Actualizar información en la interfaz
            res_text = f"📐 {w}×{h}px | {dpi_x:.0f} DPI | {width_cm:.1f}×{height_cm:.1f}cm | {file_size:.1f}MB"
            self.image_res_label.setText(res_text)

            # Guardar información para uso posterior
            self.image_info = {
                'width_px': w,
                'height_px': h,
                'dpi_x': dpi_x,
                'dpi_y': dpi_y,
                'width_cm': width_cm,
                'height_cm': height_cm,
                'file_size_mb': file_size,
                'file_path': file_path
            }

            print(f"📊 Información de imagen detectada:")
            print(f"   Dimensiones: {w}×{h} px")
            print(f"   DPI: {dpi_x}×{dpi_y}")
            print(f"   Tamaño físico: {width_cm:.1f}×{height_cm:.1f} cm")
            print(f"   Archivo: {file_size:.1f} MB")

        except Exception as e:
            print(f"⚠️ Error detectando resolución: {e}")
            self.image_res_label.setText("📐 Resolución no detectada")
            self.image_info = None

            expanded_height = paper_height_px + total_margin

            orig_h, orig_w = img.shape

            print_area_y = margin_px
            print_area_w = paper_width_px
            print_area_h = paper_height_px

            scale_x = print_area_w / orig_w if orig_w > 0 else 1.0
            scale_y = print_area_h / orig_h if orig_h > 0 else 1.0

            new_h = int(orig_h * scale)

            else:
                img_resized = img
                new_w, new_h = orig_w, orig_h

            start_x = print_area_x + (print_area_w - new_w) // 2
            start_y = print_area_y + (print_area_h - new_h) // 2

    def display_original(self):

            h, w, ch = image_rgb.shape

            # Crear un QPixmap y asignarlo al QLabel.

        except Exception as e:

            self.original_label.setText(f"Error al mostrar imagen:\n{str(e)}")

    def generate_channel_pixmap(self, channel):
        """
        ACTUALIZADO: Genera un pixmap del canal individual APLICANDO EL HALFTONE.
        """
        if channel not in self.channel_arrays:
            return QtGui.QPixmap()

        gray_channel_np = self.channel_arrays[channel]
        threshold = self.channel_thresholds.get(channel, 128)
        _, thresholded_channel = cv2.threshold(gray_channel_np, threshold, 255, cv2.THRESH_BINARY)

        scale, shape, angles = self._get_halftone_params()
        
        halftoned_mask = self.generate_quick_halftone(
            thresholded_channel, 
            angles.get(channel, 0), 
            scale
        )

        # Verificar que los datos son correctos
        unique_vals = len(np.unique(halftoned_mask))
        print(f"📊 Halftone {channel}: valores únicos = {unique_vals}, rango = {halftoned_mask.min()}-{halftoned_mask.max()}")

        if unique_vals > 1:
            print(f"✅ Halftone con ángulo {angles.get(channel, 0)}° aplicado al canal {channel}")
        else:
            print(f"⚠️ Halftone {channel} sin variación, revisa parámetros")

        h, w = halftoned_mask.shape
        rgba_data = np.zeros((h, w, 4), dtype=np.uint8)
        target_color = self.channel_colors[channel]
        r, g, b = target_color.red(), target_color.green(), target_color.blue()
        
        # Pinta los puntos (donde la máscara es negra) con el color de la tinta
        ink_mask = halftoned_mask == 0
        rgba_data[ink_mask] = [r, g, b, 255]
        
        q_image = QtGui.QImage(rgba_data.data, w, h, 4 * w, QtGui.QImage.Format_ARGB32)
        return QtGui.QPixmap.fromImage(q_image)
    
    def adjust_halftone_with_morphology(self, halftone_data, value):
        """Aplica la operación morfológica basándose en un valor (0-255)."""
        if value == 128:
            return halftone_data
        intensity = int(abs(value - 128) / 32) + 1
        kernel = np.ones((intensity, intensity), np.uint8)
        if value > 128:
            return cv2.erode(halftone_data, kernel, iterations=1)
        else:
            return cv2.dilate(halftone_data, kernel, iterations=1)

    def generate_halftone_pattern(self, channel_data, threshold, channel_name):
        """
        Halftones equilibrados para simulación natural
        """
        try:
            print(f"🎯 Generando halftone equilibrado para {channel_name}")
            
            # Normalizar datos del canal
            normalized_data = channel_data.astype(np.float32) / 255.0
            threshold_norm = threshold / 255.0
            
            # Crear patrón de puntos equilibrado
            height, width = channel_data.shape
            dot_size = self.get_halftone_dot_size(channel_name)
            angle = self.get_halftone_angle(channel_name)
            
            # Crear grilla de coordenadas con rotación
            y, x = np.mgrid[0:height, 0:width]
            
            # Aplicar rotación
            angle_rad = np.radians(angle)
            x_rot = x * np.cos(angle_rad) - y * np.sin(angle_rad)
            y_rot = x * np.sin(angle_rad) + y * np.cos(angle_rad)
            
            # Crear patrón de puntos usando coordenadas rotadas
            pattern_x = (x_rot % dot_size) - dot_size/2
            pattern_y = (y_rot % dot_size) - dot_size/2
            
            # Distancia al centro del punto
            distance = np.sqrt(pattern_x**2 + pattern_y**2)
            
            # Normalizar el patrón de forma equilibrada
            max_distance = dot_size / 2
            pattern = np.clip(distance / max_distance, 0.0, 1.0)
            
            # Aplicar función de transferencia para puntos más naturales
            pattern = np.power(pattern, 0.8)  # Menos agresivo que 0.7
            
            # Aplicar umbral de intensidad equilibrado
            intensity_adjusted = np.clip(
                (normalized_data - threshold_norm) * 2.5 + 0.2,  # Menos agresivo que 3.0
                0.0, 1.0
            )
            
            # Combinar patrón con intensidad
            halftone_mask = (pattern < intensity_adjusted).astype(np.float32)
            
            # Suavizado muy sutil
            try:
                from scipy import ndimage
                halftone_mask = ndimage.gaussian_filter(halftone_mask, sigma=0.3)  # Menos suavizado
            except ImportError:
                pass
            
            print(f"✅ Halftone equilibrado para {channel_name} - densidad: {np.mean(halftone_mask):.3f}")
            return halftone_mask
            
        except Exception as e:
            print(f"⚠️ Error en halftone equilibrado: {e}")
            # Fallback mejorado
            normalized_data = channel_data.astype(np.float32) / 255.0
            threshold_norm = threshold / 255.0
            return np.clip((normalized_data - threshold_norm) * 1.8 + 0.4, 0.0, 1.0)

    def get_halftone_angle(self, channel_name):
        """
        Ángulos estándar para evitar interferencias (moiré)
        """
        angles = {
            'C': 15,   # Cian: 15°
            'M': 75,   # Magenta: 75°
            'Y': 0,    # Amarillo: 0°
            'K': 45,   # Negro: 45°
            'W': 30    # Blanco: 30°
        }
        return angles.get(channel_name, 15)

    def create_dot_pattern(self, width, height, dot_size, angle):
        """
        Crea patrón de puntos para halftone
        """
        # Crear grilla de coordenadas
        x = np.arange(width)
        y = np.arange(height)
        X, Y = np.meshgrid(x, y)
        
        # Aplicar rotación
        angle_rad = np.radians(angle)
        X_rot = X * np.cos(angle_rad) - Y * np.sin(angle_rad)
        Y_rot = X * np.sin(angle_rad) + Y * np.cos(angle_rad)
        
        # Crear patrón de puntos circulares
        grid_x = (X_rot % dot_size) - dot_size/2
        grid_y = (Y_rot % dot_size) - dot_size/2
        
        # Distancia al centro del punto
        distance = np.sqrt(grid_x**2 + grid_y**2)
        
        # Normalizar (0.0 = centro del punto, 1.0 = borde)
        pattern = distance / (dot_size/2)
        pattern = np.clip(pattern, 0.0, 1.0)
        
        return pattern

    def apply_halftone_absorption(self, canvas, halftone_mask, channel_name):
        """
        Aplica absorción CMYK pura con patrón de halftone
        """
        if channel_name == 'W':
            # Blanco: aplicar sobre áreas con tinta
            white_areas = halftone_mask > 0.1
            canvas[white_areas, 0] = 1.0
            canvas[white_areas, 1] = 1.0
            canvas[white_areas, 2] = 1.0
            
        elif channel_name == 'C':
            # CIAN PURO: absorbe SOLO rojo con halftone
            canvas[:, :, 0] *= (1.0 - halftone_mask * 0.70)
            # Verde y azul quedan intactos
            
        elif channel_name == 'M':
            # MAGENTA PURO: absorbe SOLO verde con halftone
            canvas[:, :, 1] *= (1.0 - halftone_mask * 0.70)
            # Rojo y azul quedan intactos
            
        elif channel_name == 'Y':
            # AMARILLO PURO: absorbe SOLO azul con halftone
            canvas[:, :, 2] *= (1.0 - halftone_mask * 0.75)
            # Rojo y verde quedan intactos
            
        elif channel_name == 'K':
            # Negro: absorbe todos los canales por igual con halftone
            absorption = halftone_mask * 0.82
            canvas[:, :, 0] *= (1.0 - absorption)
            canvas[:, :, 1] *= (1.0 - absorption)
            canvas[:, :, 2] *= (1.0 - absorption)
        
        return canvas
    
    def apply_threshold_to_channel(self, halftone_data, channel_name):
        """
        Aplica una operación morfológica (erosión/dilatación) al patrón
        de halftone para simular un cambio de umbral en tiempo real.
        """
        # Si no hay un canal seleccionado o no es el canal correcto, devuelve los datos sin modificar.
        if self.current_channel is None or self.current_channel != channel_name:
            return halftone_data
        
        # Lee el valor del slider (0-255)
        value = self.threshold_slider.value()

        # El valor central (128) no aplica ningún efecto.
        if value == 128:
            return halftone_data

        # Calcula la intensidad del efecto (qué tan grandes serán los cambios)
        intensity = int(abs(value - 128) / 32) + 1
        kernel = np.ones((intensity, intensity), np.uint8)

        # En la imagen halftone, los puntos son negros (0) y el fondo blanco (255)
        if value > 128:
            # Para hacer los puntos negros MÁS GRANDES, debemos EROSIONAR el fondo blanco.
            print(f"🔎 Aplicando EROSIÓN al canal {channel_name} con intensidad {intensity}")
            adjusted_halftone = cv2.erode(halftone_data, kernel, iterations=1)
        else: # value < 128
            # Para hacer los puntos negros MÁS PEQUEÑOS, debemos DILATAR el fondo blanco.
            print(f"🔎 Aplicando DILATACIÓN al canal {channel_name} con intensidad {intensity}")
            adjusted_halftone = cv2.dilate(halftone_data, kernel, iterations=1)
            
        return adjusted_halftone

            if shape_name == 'Difusión de Error (Floyd-Steinberg)':
                return (apply_floyd_steinberg_halftone(channel_data) * 255).astype(np.uint8)

            # Lógica de generación de patrones (esta parte ya está correcta)
            angle = {'C': 15, 'M': 75, 'Y': 0, 'K': 45, 'W': 90}.get(channel_name, 0)
            center = scale / 2
            y, x = np.ogrid[:scale, :scale]
            if shape_name == 'Línea':
                screen = (y / scale) * 255
            elif shape_name == 'Elipse':
                screen = np.sqrt(((x - center) / 1.5)**2 + (y - center)**2) / center * 255
            elif shape_name == 'Diamante':
                screen = (np.abs(x - center) + np.abs(y - center)) / scale * 255
            else: # Redonda
                screen = np.sqrt((x - center)**2 + (y - center)**2) / center * 255
            
            dot_screen = np.clip(screen, 0, 255).astype(np.uint8)
            
            center_tuple = (center - 0.5, center - 0.5)
            rotation_matrix = cv2.getRotationMatrix2D(center_tuple, angle, 1.0)
            rotated_screen = cv2.warpAffine(dot_screen, rotation_matrix, (scale, scale), flags=cv2.INTER_NEAREST)
            
            h, w = channel_data.shape[:2]
            tiled_screen = np.tile(rotated_screen, (int(np.ceil(h/scale)), int(np.ceil(w/scale))))
            final_screen = tiled_screen[:h, :w]
            
            # --- ¡LA CORRECCIÓN FUNDAMENTAL ESTÁ AQUÍ! ---
            # El channel_data viene como (0=negro, 255=blanco).
            # Donde el valor del canal sea MENOR que el valor de la pantalla, ponemos tinta (255).
            halftone_result = np.where(channel_data < final_screen, 255, 0).astype(np.uint8)
            
            return halftone_result

        except Exception as e:
            print(f"❌ ERROR CRÍTICO DENTRO DE generate_halftone_from_channel_data: {e}")
            import traceback
            traceback.print_exc()
            return np.zeros(channel_data.shape, dtype=np.uint8)
    
    def _generate_channel_caches(self, source_image):
        """
        Función auxiliar DEFINITIVA. Procesa una imagen aplicando todas las
        correcciones de calidad para generar los canales y halftones.
        """
        print(f"DEBUG: Generando cachés para imagen de tamaño {source_image.shape}.")
        
        image_to_process = source_image.copy()

        # --- Mejora 1: Limpieza de Márgenes Blancos (Correcto) ---
        white_clamp_threshold = 254
        gray_version_for_margins = cv2.cvtColor(image_to_process, cv2.COLOR_BGR2GRAY)
        image_to_process[gray_version_for_margins >= white_clamp_threshold] = [255, 255, 255]
        
        # --- Mejora Opcional: Aumento de Viveza (Correcto) ---
        print("DEBUG: Aplicando compensación de saturación...")
        hsv_image = cv2.cvtColor(image_to_process, cv2.COLOR_BGR2HSV)
        saturation_factor = 1.1 # Puedes ajustar este "dial"
        hsv_image[:, :, 1] = np.clip(hsv_image[:, :, 1].astype(np.float32) * saturation_factor, 0, 255).astype(np.uint8)
        image_to_process = cv2.cvtColor(hsv_image, cv2.COLOR_HSV2BGR)

        # --- PROCESAMIENTO PRINCIPAL ---

        # a. Separación CMYK Profesional con Perfiles ICC
        # Esta llamada ya es correcta
        channel_arrays = self.convert_rgb_to_cmyk_professional(image_to_process)

        if not channel_arrays:
            return {}, {}, {}
            
        # Extraemos los canales para la limpieza. Es importante que estén en el rango 0.0-1.0
        # La función de conversión ya los devuelve en 0-255, así que los normalizamos.
        C_uint8, M_uint8, Y_uint8, K_uint8 = channel_arrays['C'], channel_arrays['M'], channel_arrays['Y'], channel_arrays['K']
        C = C_uint8.astype(np.float32) / 255.0
        M = M_uint8.astype(np.float32) / 255.0
        Y = Y_uint8.astype(np.float32) / 255.0
        K = K_uint8.astype(np.float32) / 255.0 # K también se normaliza

        # --- Mejora 2: Limpieza SELECTIVA de Altas Luces (¡LA PIEZA QUE FALTABA!) ---
        print("DEBUG: Aplicando limpieza selectiva de altas luces...")
        # Creamos una versión RGB de la imagen para medir el brillo (ya está en float)
        rgb_float_for_luminance = cv2.cvtColor(image_to_process, cv2.COLOR_BGR2RGB).astype(np.float32) / 255.0
        grayscale_luminance = cv2.cvtColor(rgb_float_for_luminance, cv2.COLOR_RGB2GRAY)
        
        highlight_threshold = 0.90
        color_residue_threshold = 0.08
        
        bright_pixels_mask = grayscale_luminance > highlight_threshold
        low_color_pixels_mask = (C < color_residue_threshold) & (M < color_residue_threshold) & (Y < color_residue_threshold)
        
        final_cleaning_mask = bright_pixels_mask & low_color_pixels_mask
        
        # Aplicamos la limpieza. Volvemos a convertir a uint8 para la caché.
        C_uint8[final_cleaning_mask] = 0
        M_uint8[final_cleaning_mask] = 0
        Y_uint8[final_cleaning_mask] = 0
        
        # Actualizamos el diccionario con los canales ya limpios
        channel_arrays['C'], channel_arrays['M'], channel_arrays['Y'] = C_uint8, M_uint8, Y_uint8
        # ---------------------------------------------------------------------

        # b. Base Blanca
        if self.white_base_cb.isChecked():
            expansion_pixels = WHITE_BASE_SETTINGS["expansion_pixels"]
            channel_arrays['W'] = generate_white_base(image_to_process, WHITE_BASE_SETTINGS["opacity_threshold"], expansion_pixels)

        # c. Generación de Halftones
        base_halftone_cache = {}
        preview_cache = {}
        for channel_name, channel_data in channel_arrays.items():
            initial_halftone = self.generate_halftone_from_channel_data(channel_name, channel_data)
            if initial_halftone is not None:
                base_halftone_cache[channel_name] = initial_halftone
                preview_cache[channel_name] = initial_halftone
                
        return channel_arrays, base_halftone_cache, preview_cache

    def _render_single_layer(self, channel_name):
        """
        Toma el halftone de UN solo canal y lo "pinta" sobre un lienzo transparente,
        devolviendo una imagen RGBA a resolución completa.
        """
        try:
            halftone_mask = self.preview_cache.get(channel_name)
            if halftone_mask is None:
                return None

            h, w = halftone_mask.shape[:2]
            print(f"DEBUG RENDER: Para el canal '{channel_name}', el diccionario contiene el color: {self.channel_colors.get(channel_name).name()}")

            ink_qcolor = self.channel_colors.get(channel_name, QtGui.QColor(0,0,0))
            ink_rgb = np.array([ink_qcolor.red(), ink_qcolor.green(), ink_qcolor.blue()], dtype=np.float32) / 255.0
            opacity = 1.0
            if channel_name == 'K': opacity = 0.98
            
            rgba_layer = np.zeros((h, w, 4), dtype=np.float32)
            rgba_layer[..., :3] = ink_rgb
            alpha_mask = (255 - halftone_mask).astype(np.float32) / 255.0
            rgba_layer[..., 3] = alpha_mask * opacity
            
            return (rgba_layer * 255).astype(np.uint8)
        except Exception as e:
            print(f"❌ Error en _render_single_layer para canal '{channel_name}': {e}")
            return None

    def fallback_rgb_to_cmyk(self, bgr_image):
        """
        Conversión RGB a CMYK básica sin perfiles ICC
        """
        print("DEBUG: Usando conversión CMYK matemática básica...")
        
        # Convertir BGR a RGB
        rgb = cv2.cvtColor(bgr_image, cv2.COLOR_BGR2RGB).astype(np.float32) / 255.0
        
        # Extraer canales
        R = rgb[:,:,0]
        G = rgb[:,:,1]
        B = rgb[:,:,2]
        
        # Calcular K (negro)
        K = 1 - np.maximum(np.maximum(R, G), B)
        
        # Evitar división por cero
        K_safe = np.where(K == 1, 0.9999, K)
        
        # Calcular CMY
        C = (1 - R - K) / (1 - K_safe)
        M = (1 - G - K) / (1 - K_safe)
        Y = (1 - B - K) / (1 - K_safe)
        
        # Convertir a 8 bits e invertir
        C = (255 - (C * 255).astype(np.uint8))
        M = (255 - (M * 255).astype(np.uint8))
        Y = (255 - (Y * 255).astype(np.uint8))
        K = (255 - (K * 255).astype(np.uint8))
        
        return {'C': C, 'M': M, 'Y': Y, 'K': K}

    def update_composite_preview(self):
        """
        Vista previa optimizada con conversión CMYK correcta.
        """
        if not hasattr(self, 'channel_arrays') or not self.channel_arrays:
            return
        if hasattr(self, 'is_preview_updating') and self.is_preview_updating:
            return

        self.is_preview_updating = True
        try:
            h, w = self.processed_image.shape[:2]
            
            # Inicializar valores CMYK
            cyan = np.zeros((h, w), dtype=np.float32)
            magenta = np.zeros((h, w), dtype=np.float32)
            yellow = np.zeros((h, w), dtype=np.float32)
            black = np.zeros((h, w), dtype=np.float32)
            
            # Obtener datos de cada canal con ajustes de umbral
            for channel_name, channel_data in self.channel_arrays.items():
                if channel_name not in ['C', 'M', 'Y', 'K']:
                    continue
                    
                # Aplicar ajuste de umbral
                current_threshold_value = self.channel_thresholds.get(channel_name, 128)
                gamma_value = (current_threshold_value / 128.0)**2.0
                gamma_value = max(0.01, gamma_value)
                gamma_lut = np.array([((i/255.0)**(1.0/gamma_value))*255 for i in np.arange(256)]).astype("uint8")
                adjusted_data = cv2.LUT(channel_data, gamma_lut)
                
                # IMPORTANTE: Invertir los valores porque en CMYK:
                # 0 = sin tinta (blanco), 255 = tinta completa
                # Pero los canales extraídos pueden estar en el formato opuesto
                inverted_data = 255 - adjusted_data
                
                # Normalizar a 0-1
                normalized = inverted_data.astype(np.float32) / 255.0
                
                # Asignar al canal correspondiente
                if channel_name == 'C':
                    cyan = normalized
                elif channel_name == 'M':
                    magenta = normalized
                elif channel_name == 'Y':
                    yellow = normalized
                elif channel_name == 'K':
                    black = normalized
            
            # Aplicar halftone si está habilitado ANTES de la conversión de color
            if self.should_show_halftones():
                # Aplicar halftone a cada canal CMYK individualmente
                cyan = self.apply_channel_halftone(cyan)
                magenta = self.apply_channel_halftone(magenta)
                yellow = self.apply_channel_halftone(yellow)
                black = self.apply_channel_halftone(black)
            
            # Conversión CMYK a RGB
            # Fórmula estándar: R = (1-C)*(1-K), G = (1-M)*(1-K), B = (1-Y)*(1-K)
            r = (1.0 - cyan) * (1.0 - black)
            g = (1.0 - magenta) * (1.0 - black)
            b = (1.0 - yellow) * (1.0 - black)
            
            # Combinar canales RGB
            rgb_image = np.stack([r, g, b], axis=-1)
            
            # Aplicar color de prenda si no es blanco
            garment_qcolor = self.garment_color_btn.palette().button().color()
            if garment_qcolor != QtGui.QColor(255, 255, 255):
                garment_rgb = np.array([
                    garment_qcolor.redF(), 
                    garment_qcolor.greenF(), 
                    garment_qcolor.blueF()
                ], dtype=np.float32)
                # Multiplicar por el color de la prenda
                rgb_image = rgb_image * garment_rgb
            
            # Convertir a uint8
            final_image_uint8 = np.clip(rgb_image * 255, 0, 255).astype(np.uint8)
            
            # Mostrar en la vista previa
            pixmap = self.convertir_cv_a_pixmap(final_image_uint8)
            
            if self.preview_pixmap_item is None:
                self.preview_scene.clear()
                self.preview_pixmap_item = self.preview_scene.addPixmap(pixmap)
                self.preview_view.fitInView(self.preview_scene.itemsBoundingRect(), QtCore.Qt.KeepAspectRatio)
            else:
                self.preview_pixmap_item.setPixmap(pixmap)

        except Exception as e:
            import traceback
            print(f"❌ ERROR en update_composite_preview: {e}")
            traceback.print_exc()
        finally:
            self.is_preview_updating = False

    def apply_channel_halftone(self, channel):
        """
        Aplica halftone con puntos circulares basados en la lineatura real.
        """
        h, w = channel.shape
        
        # Obtener DPI y LPI de los campos correctos
        dpi = self.custom_dpi.value()
        
        # Obtener el valor LPI del diccionario
        lpi_key = self.lpi_combo.currentText()
        lpi = LPI_VALUES.get(lpi_key, 45)  # 45 como valor por defecto
        
        cell_size = int(dpi / lpi)
        
        # Asegurar un tamaño mínimo
        if cell_size < 3:
            cell_size = 3
        
        # Crear salida en blanco
        output = np.zeros_like(channel)
        
        # Pre-calcular máscara circular
        center = cell_size / 2.0
        y_grid, x_grid = np.ogrid[:cell_size, :cell_size]
        dist_from_center = np.sqrt((x_grid - center)**2 + (y_grid - center)**2)
        
        # Procesar por celdas
        for y in range(0, h - cell_size + 1, cell_size):
            for x in range(0, w - cell_size + 1, cell_size):
                # Valor promedio de la celda
                avg_value = np.mean(channel[y:y+cell_size, x:x+cell_size])
                
                # Umbral mínimo para mostrar puntos
                if avg_value > 0.01:
                    # Radio proporcional a la intensidad
                    max_radius = cell_size / 2.0
                    radius = max_radius * np.sqrt(avg_value)
                    
                    # Crear máscara circular
                    mask = dist_from_center <= radius
                    
                    # Aplicar el punto con la intensidad completa
                    output[y:y+cell_size, x:x+cell_size][mask] = avg_value
        
        # Manejar bordes
        if h % cell_size != 0:
            for x in range(0, w - cell_size + 1, cell_size):
                y = (h // cell_size) * cell_size
                avg_value = np.mean(channel[y:h, x:x+cell_size])
                if avg_value > 0.01:
                    output[y:h, x:x+cell_size] = avg_value
        
        if w % cell_size != 0:
            for y in range(0, h - cell_size + 1, cell_size):
                x = (w // cell_size) * cell_size
                avg_value = np.mean(channel[y:y+cell_size, x:w])
                if avg_value > 0.01:
                    output[y:y+cell_size, x:w] = avg_value
        
        return output

    def generate_realistic_halftone(self, channel_data, angle=45, dpi=150, dot_shape='round'):
        """
        Genera un halftone realista con el ángulo especificado.
        """
        h, w = channel_data.shape
        
        # Calcular el tamaño de la celda basado en el DPI
        lpi = dpi / 2.5  # Lines per inch (típicamente DPI/2.5)
        cell_size = int(dpi / lpi)
        if cell_size < 4:
            cell_size = 4
        
        # Crear imagen de salida
        halftone = np.ones((h, w), dtype=np.uint8) * 255
        
        # Crear una grilla rotada
        angle_rad = np.radians(angle)
        
        # Calcular el paso de la grilla considerando la rotación
        step = cell_size
        
        # Generar puntos de la grilla con el ángulo especificado
        # Necesitamos cubrir un área más grande para la rotación
        max_dim = int(np.sqrt(h*h + w*w))
        x_points = np.arange(-max_dim, max_dim, step)
        y_points = np.arange(-max_dim, max_dim, step)
        
        for x_base in x_points:
            for y_base in y_points:
                # Rotar el punto
                x_rot = x_base * np.cos(angle_rad) - y_base * np.sin(angle_rad) + w/2
                y_rot = x_base * np.sin(angle_rad) + y_base * np.cos(angle_rad) + h/2
                
                # Convertir a coordenadas enteras
                x_center = int(x_rot)
                y_center = int(y_rot)
                
                # Verificar si está dentro de la imagen
                if 0 <= x_center < w and 0 <= y_center < h:
                    # Calcular el área de muestreo
                    x_start = max(0, x_center - cell_size//2)
                    x_end = min(w, x_center + cell_size//2)
                    y_start = max(0, y_center - cell_size//2)
                    y_end = min(h, y_center + cell_size//2)
                    
                    if x_start < x_end and y_start < y_end:
                        # Obtener el valor promedio en esta celda
                        area_value = np.mean(channel_data[y_start:y_end, x_start:x_end])
                        
                        # Calcular el radio del punto basado en la intensidad
                        # Invertir: valores oscuros = puntos grandes
                        intensity = 1.0 - (area_value / 255.0)
                        radius = int(cell_size * 0.9 * np.sqrt(intensity))
                        
                        if radius > 0:
                            # Dibujar el punto
                            cv2.circle(halftone, (x_center, y_center), radius, 0, -1)
        
        return halftone

    def apply_fast_halftone_effect(self, rgb_image):
        """
        Aplica un efecto de halftone rápido para la vista previa.
        No es una simulación real pero da una idea visual.
        """
        h, w = rgb_image.shape[:2]
        
        # Tamaño de celda para el efecto
        cell_size = 4
        
        # Crear una copia para modificar
        output = rgb_image.copy()
        
        # Aplicar efecto de halftone simple
        for y in range(0, h - cell_size, cell_size):
            for x in range(0, w - cell_size, cell_size):
                # Obtener el valor promedio de la celda
                cell_avg = np.mean(rgb_image[y:y+cell_size, x:x+cell_size], axis=(0, 1))
                
                # Calcular el tamaño del punto basado en la intensidad
                intensity = np.mean(cell_avg)
                dot_size = int(cell_size * intensity)
                
                # Llenar la celda con blanco
                output[y:y+cell_size, x:x+cell_size] = 1.0
                
                # Dibujar un punto del color promedio
                if dot_size > 0:
                    center_y = y + cell_size // 2
                    center_x = x + cell_size // 2
                    y1 = max(0, center_y - dot_size // 2)
                    y2 = min(h, center_y + dot_size // 2)
                    x1 = max(0, center_x - dot_size // 2)
                    x2 = min(w, center_x + dot_size // 2)
                    output[y1:y2, x1:x2] = cell_avg
        
        return output

    def generate_quick_halftone(self, channel_data, angle, scale):
        """
        CORREGIDO: Genera un patrón de halftone clásico con puntos circulares.
        Esta versión crea puntos cuyo tamaño depende de la intensidad de la imagen y 
        cuya frecuencia (LPI) depende del 'scale' correctamente.
        """
        try:
            h, w = channel_data.shape
            
            # 1. Crear una grilla de coordenadas
            y_coords, x_coords = np.mgrid[0:h, 0:w].astype(np.float32)

            # 2. Rotar la grilla según el ángulo del canal
            angle_rad = np.radians(angle)
            cos_a, sin_a = np.cos(angle_rad), np.sin(angle_rad)
            x_rot = x_coords * cos_a - y_coords * sin_a
            y_rot = x_coords * sin_a + y_coords * cos_a

            # 3. Crear el patrón de puntos usando la grilla rotada y la escala
            # El operador módulo (%) crea una grilla repetitiva (la base del halftone)
            dot_pattern = np.sqrt(
                ((x_rot % scale) - scale / 2) ** 2 +
                ((y_rot % scale) - scale / 2) ** 2
            )
            
            # 4. Normalizar el patrón (0 = centro del punto, 1 = borde)
            max_dist = scale / 2 * np.sqrt(2)
            dot_pattern_norm = np.clip(dot_pattern / (scale / 2), 0, 1)

            # 5. Normalizar la imagen del canal (0 = negro, 1 = blanco)
            image_norm = channel_data.astype(np.float32) / 255.0
            
            # 6. Generar el halftone: un píxel se convierte en punto si su
            # intensidad en la imagen es mayor que su valor en el patrón de puntos.
            halftone = (image_norm > dot_pattern_norm).astype(np.uint8) * 255
            
            # Devolvemos la máscara invertida: los puntos son negros (0) y el fondo blanco (255)
            return cv2.bitwise_not(halftone)

        except Exception as e:
            print(f"❌ Error en generate_quick_halftone: {e}")
            # Si falla, devuelve la imagen original sin procesar
            return channel_data
        
    def create_circular_dots(dot_pattern):
        """Crear puntos más circulares"""
        return np.sin(dot_pattern * np.pi) * 0.8

    def generate_simple_halftone_pattern(self, channel_data, threshold, channel_name):
        """
        Genera halftones simples pero efectivos - CON UMBRAL FUNCIONAL
        """
        try:
            print(f"🎯 Generando halftone para {channel_name} con umbral {threshold}")
            
            # Normalizar datos del canal
            normalized_data = channel_data.astype(np.float32) / 255.0
            
            # Crear patrón de puntos
            height, width = channel_data.shape
            dot_size = self.get_simple_dot_size(channel_name)
            angle = self.get_halftone_angle(channel_name)
            
            # Crear grilla de coordenadas
            y, x = np.mgrid[0:height, 0:width]
            
            # Aplicar rotación
            angle_rad = np.radians(angle)
            x_rot = x * np.cos(angle_rad) - y * np.sin(angle_rad)
            y_rot = x * np.sin(angle_rad) + y * np.cos(angle_rad)
            
            # Crear patrón de puntos
            pattern_x = (x_rot % dot_size) - dot_size/2
            pattern_y = (y_rot % dot_size) - dot_size/2
            distance = np.sqrt(pattern_x**2 + pattern_y**2)
            
            # Normalizar patrón
            max_distance = dot_size / 2
            pattern = np.clip(distance / max_distance, 0.0, 1.0)
            
            # APLICAR UMBRAL - El umbral modifica el tamaño de los puntos
            if threshold != 64 and channel_name == 'K':  # Para canal K
                default_threshold = 64
            elif threshold != 128:  # Para otros canales
                default_threshold = 128
            else:
                default_threshold = threshold
                
            if threshold != default_threshold:
                # EFECTO MÁS DRAMÁTICO Y VISIBLE
                if threshold > default_threshold:
                    # Umbral alto = puntos MUY pequeños (imagen mucho más clara)
                    intensity_factor = 0.3 + (threshold - default_threshold) / 255.0 * 1.2
                    adjusted_intensity = np.power(normalized_data, intensity_factor)
                else:
                    # Umbral bajo = puntos MUY grandes (imagen mucho más oscura)  
                    intensity_factor = 2.5 - threshold / 255.0 * 1.8
                    adjusted_intensity = np.power(normalized_data, 1.0/intensity_factor)
                    
                print(f"   🎯 Umbral {threshold} aplicado - factor DRAMÁTICO: {intensity_factor:.2f}")
            else:
                adjusted_intensity = normalized_data
            
            # Crear máscara de halftones con intensidad ajustada
            halftone_mask = (pattern < adjusted_intensity).astype(np.float32)
            
            # Suavizado mínimo
            try:
                from scipy import ndimage
                halftone_mask = ndimage.gaussian_filter(halftone_mask, sigma=0.2)
            except ImportError:
                pass
            
            print(f"✅ Halftone para {channel_name} - densidad: {np.mean(halftone_mask):.3f}")
            return halftone_mask
        
        except Exception as e:
            print(f"⚠️ Error en halftone: {e}")
            # Fallback: usar datos originales normalizados
            return channel_data.astype(np.float32) / 255.0

    def auto_adjust_based_on_image(self, canvas, selected_channels):
        """
        Ajustes automáticos basados en el contenido de la imagen
        """
        # Analizar el contenido de la imagen
        avg_brightness = np.mean(canvas)
        avg_saturation = np.std(canvas)
        
        print(f"📊 Análisis automático - Brillo: {avg_brightness:.3f}, Saturación: {avg_saturation:.3f}")
        
        # Ajustes automáticos basados en contenido
        if avg_brightness < 0.3:
            # Imagen oscura - aumentar brillo
            canvas = np.clip(canvas + 0.1, 0.0, 1.0)
            print("🔆 Imagen oscura detectada - aumentando brillo")
        elif avg_brightness > 0.8:
            # Imagen muy clara - reducir ligeramente
            canvas = np.clip(canvas * 0.95, 0.0, 1.0)
            print("🔅 Imagen muy clara detectada - reduciendo brillo")
        
        if avg_saturation > 0.3:
            # Imagen muy contrastada - suavizar ligeramente
            canvas = np.clip((canvas - 0.5) * 0.9 + 0.5, 0.0, 1.0)
            print("🎨 Alto contraste detectado - suavizando")
        
        return canvas

    def show_detailed_preview(self):
        """
        Genera y muestra una ventana emergente con una porción de la vista previa
        a resolución 1:1, aplicando la MISMA lógica de renderizado que la vista principal
        para garantizar la consistencia y seguridad de los datos.
        """
        if not hasattr(self, 'channel_arrays') or not self.channel_arrays:
            QtWidgets.QMessageBox.warning(self, "Aviso", "Primero debes procesar una imagen.")
            return

        try:
            # --- Recortar el área de interés ---
            crop_size = 400
            h, w = self.processed_image.shape[:2]
            start_x = max(0, (w - crop_size) // 2)
            start_y = max(0, (h - crop_size) // 2)
            end_x = start_x + crop_size
            end_y = start_y + crop_size

            channel_crops = {
                name: data[start_y:end_y, start_x:end_x] 
                for name, data in self.channel_arrays.items()
            }
            if hasattr(self, 'content_mask'):
                content_mask_crop = self.content_mask[start_y:end_y, start_x:end_x]
            else:
                content_mask_crop = None

            # --- Lógica de renderizado ---
            garment_qcolor = self.garment_color_btn.palette().button().color()
            # Obtenemos las dimensiones reales del recorte por si está en un borde
            crop_h, crop_w = channel_crops[list(channel_crops.keys())[0]].shape[:2]
            canvas = np.full((crop_h, crop_w, 3),
                              (garment_qcolor.redF(), garment_qcolor.greenF(), garment_qcolor.blueF()),
                              dtype=np.float32)

            order = self.get_channel_order_from_ui()

            for channel_name in order:
                if channel_name not in channel_crops:
                    continue

                # --- ¡CORRECCIÓN CLAVE Y FINAL! ---
                # Usamos .copy() aquí para crear una fotocopia del recorte antes de modificarlo.
                channel_crop_data = channel_crops[channel_name].copy()

                # 1. APLICAMOS LA LÓGICA DE LIMPIEZA
                noise_threshold = 8
                channel_crop_data[channel_crop_data < noise_threshold] = 0
                if content_mask_crop is not None and content_mask_crop.shape == channel_crop_data.shape:
                    channel_crop_data[content_mask_crop == 255] = 0
                
                # 2. APLICAMOS LA LÓGICA DE UMBRAL (GAMMA)
                current_threshold_value = self.channel_thresholds.get(channel_name, 128)
                gamma_value = (current_threshold_value / 128.0)**2.0
                gamma_value = max(0.01, gamma_value)
                gamma_lut = np.array([((i/255.0)**(1.0/gamma_value))*255 for i in np.arange(256)]).astype("uint8")
                adjusted_data = cv2.LUT(channel_crop_data, gamma_lut)

                # 3. GENERAMOS EL HALFTONE (le indicamos que no es para preview para usar el DPI real)
                halftone_mask = self.generate_halftone_from_channel_data(channel_name, adjusted_data, is_for_preview=False)
                
                # 4. MEZCLAMOS LOS COLORES
                ink_mask = (255 - halftone_mask).astype(np.float32) / 255.0
                ink_mask = ink_mask[..., np.newaxis]
                ink_qcolor = self.channel_colors.get(channel_name, QtGui.QColor("black"))
                ink_rgb = np.array([ink_qcolor.redF(), ink_qcolor.greenF(), ink_qcolor.blueF()], dtype=np.float32)

                if channel_name == 'W':
                    canvas = (ink_rgb * ink_mask) + (canvas * (1 - ink_mask))
                else:
                    canvas = canvas * (1.0 - (ink_mask * (1.0 - ink_rgb)))

            # --- Mostrar el resultado ---
            final_image_uint8 = np.clip(canvas * 255, 0, 255).astype(np.uint8)
            pixmap = self.convertir_cv_a_pixmap(final_image_uint8)

            dialog = QtWidgets.QDialog(self)
            dialog.setWindowTitle("Vista de Detalle 1:1")
            layout = QtWidgets.QVBoxLayout()
            label = QtWidgets.QLabel()
            label.setPixmap(pixmap)
            layout.addWidget(label)
            dialog.setLayout(layout)
            dialog.exec_()

    def find_nearest_non_multiple(self, target_lpi, mesh_count):
        """Encontrar LPI cercano que no cause moiré"""
        # Verificar si es múltiplo problemático
        if mesh_count % target_lpi == 0 or target_lpi % (mesh_count // 10) == 0:
            # Buscar alternativas cercanas
            alternatives = []
            for offset in range(1, 6):
                for candidate in [target_lpi - offset, target_lpi + offset]:
                    if candidate > 0 and mesh_count % candidate != 0:
                        alternatives.append(candidate)

            return min(alternatives) if alternatives else target_lpi

        return target_lpi

    def check_moire_risk(self):
        """Verificar riesgo de moiré"""
        try:
            mesh = self.moire_mesh.value()
            lpi = self.moire_lpi.value()
            if lpi == 0: return # Avoid division by zero

            # Análisis de moiré
            is_exact_multiple = (mesh % lpi) == 0
            is_close_multiple = any((mesh % (lpi + i)) == 0 for i in range(-2, 3))
            is_harmonic = (lpi % (mesh // 10)) == 0 if (mesh // 10) > 0 else False

            risk_level = "BAJO"
            risk_color = "#4CAF50"

            if is_exact_multiple:
                risk_level = "CRÍTICO"
                risk_color = "#F44336"
            elif is_close_multiple or is_harmonic:
                risk_level = "ALTO"
                risk_color = "#FF9800"

            analysis_text = f"""
            <div style="background: {risk_color}; color: white; padding: 8px; border-radius: 4px; margin-bottom: 10px;">
                <b>RIESGO DE MOIRÉ: {risk_level}</b>
            </div>

            <b>📊 Análisis técnico:</b><br>
            • Malla {mesh} ÷ LPI {lpi} = {mesh/lpi:.2f}<br>
            • ¿Múltiplo exacto? {'❌ SÍ' if is_exact_multiple else '✅ NO'}<br>
            • ¿Múltiplo cercano? {'⚠️ SÍ' if is_close_multiple else '✅ NO'}<br>
            • ¿Armónico problemático? {'⚠️ SÍ' if is_harmonic else '✅ NO'}<br>
            """

            if risk_level != "BAJO":
                analysis_text += f"""<br><b style="color: {risk_color};">⚠️ RECOMENDACIÓN:</b> Cambiar LPI para evitar patrones de interferencia."""

            self.moire_result_label.setText(analysis_text)

            # Generar alternativas seguras
            safe_alternatives = []
            for candidate in range(max(10, lpi-5), lpi+6):
                if candidate != lpi and candidate > 0 and mesh % candidate != 0:
                    safe_alternatives.append(candidate)

            if safe_alternatives:
                alternatives_text = f"""
                <b>💡 LPI alternativos seguros:</b><br>
                {', '.join(map(str, safe_alternatives[:8]))}

                <br><br><b>🎯 Recomendación principal:</b> {safe_alternatives[len(safe_alternatives)//2]} LPI
                """
            else:
                alternatives_text = "No se encontraron alternativas en el rango cercano."

            self.alternatives_label.setText(alternatives_text)

        except Exception as e:
            self.moire_result_label.setText(f"Error: {str(e)}")

    def apply_calculated_lpi(self, dialog):
        """Aplicar LPI calculado al proyecto actual"""
        try:
            # Obtener el LPI calculado de la pestaña activa
            optimal_lpi = None

            # Desde pestaña de malla
            if hasattr(self, 'lpi_result_label'):
                text = self.lpi_result_label.text()
                import re
                match = re.search(r'LPI Calculado: (\d+)', text)
                if match:
                    optimal_lpi = int(match.group(1))

            if optimal_lpi:
                # Buscar en LPI_VALUES la opción más cercana
                closest_option = None
                min_diff = float('inf')

                for lpi_option in LPI_VALUES.keys():
                    lpi_num = int(lpi_option.split()[0])
                    diff = abs(lpi_num - optimal_lpi)
                    if diff < min_diff:
                        min_diff = diff
                        closest_option = lpi_option

                if closest_option:
                    # Aplicar al combo box principal
                    self.lpi_combo.setCurrentText(closest_option)
                    dialog.accept()

                    msg = f"✅ LPI aplicado al proyecto:\n{closest_option}\n\nPuedes procesar la imagen con esta configuración optimizada."
                    QtWidgets.QMessageBox.information(self, "LPI Aplicado", msg)
                    self.status_bar.showMessage(f"✅ LPI optimizado aplicado: {closest_option}", 5000)
                else:
                    QtWidgets.QMessageBox.warning(self, "Aviso", "No se encontró una opción LPI compatible en el sistema.")
            else:
                QtWidgets.QMessageBox.warning(self, "Aviso", "No se pudo obtener el LPI calculado.")

        except Exception as e:
            QtWidgets.QMessageBox.critical(self, "Error", f"Error al aplicar LPI:\n{str(e)}")

    def regenerate_halftone_for_channel(self, channel_name):
        """
        Toma los datos originales de un canal, calcula la escala físicamente precisa
        y vuelve a generar el halftone para ese canal, actualizando el caché.
        """
        # Se asegura de que existan los datos necesarios antes de proceder.
        if channel_name not in self.channel_arrays or self.processed_image is None:
            return 0 # Devuelve 0 si no se puede calcular

        # 1. Obtener la escala dinámica físicamente precisa y el tamaño del punto.
        # Llama a la otra función auxiliar que ya tienes.
        scale, dot_microns = self._calculate_dynamic_scale(channel_name)
        
        # 2. Obtener los datos del canal original (la separación en escala de grises).
        original_channel_data = self.channel_arrays[channel_name]
        
        # 3. Generar el nuevo halftone con la escala correcta.
        angle = {'C': 15, 'M': 75, 'Y': 0, 'K': 45, 'W': 90}.get(channel_name, 0)
        new_halftone = self.generate_quick_halftone(original_channel_data, angle, scale)

        # 4. Actualizar el caché de la vista previa con el nuevo halftone.
        self.preview_cache[channel_name] = new_halftone
        
        # 5. Devolver el tamaño del punto para que se pueda mostrar en la etiqueta de la UI.
        return dot_microns
    
    def _calculate_dynamic_scale(self, channel_name):
        """
        Calcula el 'scale' (tamaño de punto en píxeles) para simular el
        tamaño físico real basado en LPI, formato y dimensiones de la imagen.
        """
        # Obtener LPI seleccionado
        try:
            lpi = int(self.lpi_combo.currentText().split()[0])
        except (ValueError, IndexError):
            lpi = 55 # Default
        
        # Si el canal es Blanco, puede tener un LPI diferente
        if channel_name == 'W':
            lpi = WHITE_BASE_SETTINGS.get('lpi', lpi)

        # 1. Calcular tamaño físico del punto en mm
        # Fórmula: (1 pulgada / LPI) * 25.4 mm/pulgada
        dot_size_mm = (1 / lpi) * 25.4

        # 2. Obtener dimensiones físicas del formato de impresión en mm
        print_format = self.get_current_print_format()
        format_width_mm = print_format["width"]

        # 3. Obtener dimensiones en píxeles de nuestra imagen procesada
        canvas_height_px, canvas_width_px = self.processed_image.shape[:2]

        # 4. Calcular el factor de escala real (píxeles por mm)
        px_per_mm = canvas_width_px / format_width_mm

        # 5. Calcular el tamaño final del punto en píxeles para la simulación
        scale_in_pixels = dot_size_mm * px_per_mm

        # Asegurarse de que la escala sea al menos 2 para evitar errores
        final_scale = max(2, int(scale_in_pixels))
        
        dot_microns = dot_size_mm * 1000
        print(f"Canal {channel_name} @ {lpi} LPI | Tamaño físico: {dot_microns:.0f} µm | Escala en Píxeles: {final_scale}")

        return final_scale, dot_microns
    
    def on_print_settings_changed(self):
        """
        Se activa al cambiar el Formato o el LPI. Regenera todos los halftones
        con la nueva escala físicamente precisa.
        """
        # Solo se ejecuta si ya hay una imagen procesada
        if not self.preview_cache or self.is_preview_updating:
            return

        print("\n🔄 Ajustes de impresión cambiaron. Regenerando todos los halftones...")
        self.is_preview_updating = True # Bloquear para evitar bucles

        last_dot_microns = 0
        for channel_name in self.preview_cache.keys():
            # Llama a la regeneración de cada canal. La función ahora devuelve el tamaño del punto
            dot_microns = self.regenerate_halftone_for_channel(channel_name)
            if channel_name != 'W': # Usualmente nos importa el de los colores
                last_dot_microns = dot_microns

        # Actualiza la etiqueta visual con el último tamaño calculado
        if last_dot_microns > 0:
            self.dot_size_label.setText(f"Tamaño de punto: {last_dot_microns:.0f} µm")

        self.is_preview_updating = False # Desbloquear
        self.update_composite_preview() # Actualizar la vista final

    def process_cmyk(self):
        """
        Genera la separación y los halftones base, aplicando corrección Gamma
        y guardando los resultados en los cachés para su edición.
        VERSIÓN MODIFICADA PARA SOPORTAR MINIATURAS.
        """
        if self.image is None:
            QtWidgets.QMessageBox.warning(self, "Aviso", "Carga una imagen primero")
            return
        try:
            self.status_bar.showMessage("Procesando separación CMYK profesional...")
            
            # --- Pre-procesamiento (sin cambios) ---
            working_image = self.image.copy()
            resolution_settings = self.get_current_resolution_settings()
            if resolution_settings["factor"] > 1.0:
                working_image = enhance_image_resolution(working_image, resolution_settings["factor"], resolution_settings["method"])
            print_format = self.get_current_print_format()
            if self.fit_format_cb.isChecked():
                working_image = resize_to_print_format(working_image, print_format, print_format["dpi_recommended"])

            # --- Creación de datos de Alta y Baja Resolución ---
            
            # 1. Guarda la imagen final a resolución completa
            self.processed_image = working_image.copy()

            print("DEBUG: Creando máscara de contenido maestra...")
            # Convertimos la imagen procesada a escala de grises
            gray_processed = cv2.cvtColor(self.processed_image, cv2.COLOR_BGR2GRAY)
            # Creamos una máscara binaria. Todo lo que no sea blanco puro (255) se vuelve negro (0).
            # El fondo blanco se vuelve 255, el contenido es más oscuro.
            _, mask = cv2.threshold(gray_processed, 254, 255, cv2.THRESH_BINARY)
            # La máscara ahora tiene el contenido en negro (0) y el fondo en blanco (255).
            self.content_mask = mask # Guardamos esta máscara para usarla en la vista previa.
            

            print("DEBUG: Creando máscara de contenido maestra...")
            # Convertimos la imagen procesada a escala de grises
            gray_processed = cv2.cvtColor(self.processed_image, cv2.COLOR_BGR2GRAY)
            
            # Creamos una máscara binaria. Todo lo que no sea blanco puro (255) se vuelve negro (0).
            # Usamos un umbral de 254 para capturar los "casi-blancos".
            _, mask = cv2.threshold(gray_processed, 254, 255, cv2.THRESH_BINARY)
            
            # La máscara ahora tiene el contenido en negro y los márgenes en blanco. La guardamos.
            self.content_mask = mask

            # --- Lógica de Limpieza de Márgenes Blancos ---
            print("DEBUG: Limpiando márgenes blancos de la imagen procesada principal...")
            white_clamp_threshold = 250  # Umbral para considerar "blanco puro"
            
            # Convierte la imagen procesada a escala de grises para encontrar los píxeles casi blancos
            gray_version = cv2.cvtColor(self.processed_image, cv2.COLOR_BGR2GRAY)
            
            # En la imagen original procesada, donde los píxeles sean casi blancos, fuérzalos a ser blanco puro.
            self.processed_image[gray_version > white_clamp_threshold] = [255, 255, 255]
            
            # 2. Crea la versión en miniatura a partir de la imagen ya procesada
            print("DEBUG: Creando miniatura con proporción correcta...")
            MAX_PREVIEW_DIMENSION = 400  # El tamaño máximo para el lado más largo de la miniatura
            h, w = self.processed_image.shape[:2]

            if h > w:
                # Si la imagen es más alta que ancha
                new_h = MAX_PREVIEW_DIMENSION
                new_w = int(w * (new_h / h)) # Calcula el ancho proporcional
            else:
                # Si la imagen es más ancha que alta (o cuadrada)
                new_w = MAX_PREVIEW_DIMENSION
                new_h = int(h * (new_w / w)) # Calcula el alto proporcional

            # Redimensiona a las nuevas dimensiones calculadas
            self.processed_image_thumb = cv2.resize(self.processed_image, (new_w, new_h), interpolation=cv2.INTER_AREA)
            print(f"DEBUG: Miniatura con proporción corregida creada. Dimensiones: {self.processed_image_thumb.shape}")

            # 3. Llama a la función auxiliar para generar los cachés de ALTA RESOLUCIÓN
            print("--- PROCESANDO DATOS EN ALTA RESOLUCIÓN ---")
            self.channel_arrays, self.base_halftone_cache, self.preview_cache = self._generate_channel_caches(self.processed_image)

            # 4. Llama a la función auxiliar para generar los cachés de BAJA RESOLUCIÓN (miniatura)
            print("--- PROCESANDO DATOS EN BAJA RESOLUCIÓN (MINIATURA) ---")
            self.channel_arrays_thumb, self.base_halftone_cache_thumb, self.preview_cache_thumb = self._generate_channel_caches(self.processed_image_thumb)
            
            print("DEBUG: Pre-renderizando capas para la composición rápida...")
            self.rendered_layer_cache = {}
            for channel in self.get_channel_order_from_ui():
                # Intentamos renderizar la capa
                rendered_layer = self._render_single_layer(channel)
                
                # Solo añadimos la capa a la caché si se pudo renderizar (no es None)
                if rendered_layer is not None:
                    self.rendered_layer_cache[channel] = rendered_layer
                    print(f"  -> Capa '{channel}' pre-renderizada con éxito.")

            # Finalmente, llama a la actualización de la composición
            self.display_original()
            self.update_composite_preview()

            # --- Limpieza y Actualización Final de la Interfaz ---
            self.channel_thresholds = self.PURE_DEFAULT_THRESHOLDS.copy()
            
            print("DEBUG: Procesamiento finalizado. Llamando a las funciones de actualización de la interfaz...")
            self.display_original()
            self.update_composite_preview() # Esta mostrará el resultado de alta calidad por defecto
            
            self.status_bar.showMessage("✅ Separación y vista previa completadas con éxito.")
            
        except Exception as e:
            error_msg = f"Error en procesamiento CMYK: {str(e)}"
            import traceback
            traceback.print_exc()
            QtWidgets.QMessageBox.critical(self, "Error", error_msg)
